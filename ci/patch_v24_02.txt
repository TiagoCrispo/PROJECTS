+import kotlin.math.max
+import kotlin.math.min
+
+/**
+ * Conservative cross-origin reconciliation for cumulative interval metrics.
+ * It divides time into one-minute buckets and keeps the largest contribution in
+ * every bucket instead of summing overlapping sources (watch + phone + app).
+ * Non-overlapping intervals are still fully accumulated.
+ */
+object HealthRecordDeduplicator {
+    data class Interval(
+        val startEpochMs: Long,
+        val endEpochMs: Long,
+        val value: Double,
+        val origin: String = "",
+    )
+
+    data class Result(
+        val value: Double,
+        val inputRecords: Int,
+        val overlapSuppressedRecords: Int,
+    )
+
+    private const val BUCKET_MS = 60_000L
+
+    fun reconcile(records: List<Interval>): Result {
+        val valid = records.filter { it.value >= 0.0 && it.endEpochMs > it.startEpochMs }
+        if (valid.isEmpty()) return Result(0.0, records.size, 0)
+
+        val bucketBest = mutableMapOf<Long, Double>()
+        valid.forEach { r ->
+            val duration = (r.endEpochMs - r.startEpochMs).toDouble()
+            var bucket = Math.floorDiv(r.startEpochMs, BUCKET_MS) * BUCKET_MS
+            while (bucket < r.endEpochMs) {
+                val bucketEnd = bucket + BUCKET_MS
+                val overlap = (min(r.endEpochMs, bucketEnd) - max(r.startEpochMs, bucket)).coerceAtLeast(0L)
+                if (overlap > 0) {
+                    val contribution = r.value * (overlap / duration)
+                    val old = bucketBest[bucket] ?: 0.0
+                    if (contribution > old) bucketBest[bucket] = contribution
+                }
+                bucket += BUCKET_MS
+            }
+        }
+
+        val suppressed = valid.count { r ->
+            valid.any { other ->
+                if (other === r || other.origin == r.origin) false
+                else {
+                    val overlap = (min(r.endEpochMs, other.endEpochMs) - max(r.startEpochMs, other.startEpochMs)).coerceAtLeast(0L)
+                    val minDuration = min(r.endEpochMs - r.startEpochMs, other.endEpochMs - other.startEpochMs).coerceAtLeast(1L)
+                    overlap.toDouble() / minDuration >= 0.80
+                }
+            }
+        }
+        return Result(bucketBest.values.sum(), records.size, suppressed)
+    }
+}
