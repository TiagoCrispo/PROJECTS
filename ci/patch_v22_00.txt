# Pure strength math helper
p=root/'app/src/main/java/com/fer/bandlab/training/StrengthMath.kt'
p.write_text('''package com.fer.bandlab.training\n\nimport com.fer.bandlab.model.StrengthSetRecord\nimport kotlin.math.round\n\n/** Deterministic training math only. No fabricated measurements. */\nobject StrengthMath {\n    fun e1rm(weightKg: Double, reps: Int): Double {\n        require(weightKg >= 0.0)\n        require(reps >= 1)\n        return if (reps <= 1) weightKg else weightKg * (1.0 + reps / 30.0)\n    }\n\n    fun suggestedWeight(latest: StrengthSetRecord, plateStepKg: Double = 0.5): Double {\n        val multiplier = when {\n            latest.reps >= 12 -> 1.05\n            latest.reps >= 8 -> 1.025\n            latest.reps <= 4 -> 0.975\n            else -> 1.0\n        }\n        return roundToPlate(latest.weightKg * multiplier, plateStepKg)\n    }\n\n    fun roundToPlate(value: Double, step: Double): Double {\n        require(step > 0.0)\n        return round(value / step) * step\n    }\n}\n''')
