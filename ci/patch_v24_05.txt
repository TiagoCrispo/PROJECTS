+@Composable
+fun OfflineRouteView(route: List<RoutePoint>, modifier: Modifier = Modifier) {
+    Box(
+        modifier = modifier.fillMaxWidth().height(190.dp).background(Surface2, RoundedCornerShape(20.dp)),
+        contentAlignment = Alignment.Center,
+    ) {
+        if (route.size < 2) {
+            Text("Sin ruta GPS suficiente", color = TextSecondary, style = MaterialTheme.typography.bodyMedium)
+            return@Box
+        }
+        val endpointColor = MaterialTheme.colorScheme.primary
+        Canvas(modifier = Modifier.fillMaxSize()) {
+            val pad = 18.dp.toPx()
+            val midLat = route.map { it.latitude }.average()
+            val lonScale = cos(Math.toRadians(midLat)).coerceAtLeast(0.2)
+            val xs = route.map { it.longitude * lonScale }
+            val ys = route.map { it.latitude }
+            val minX = xs.minOrNull() ?: return@Canvas; val maxX = xs.maxOrNull() ?: return@Canvas
+            val minY = ys.minOrNull() ?: return@Canvas; val maxY = ys.maxOrNull() ?: return@Canvas
+            val dx = (maxX - minX).takeIf { it > 1e-9 } ?: 1e-9
+            val dy = (maxY - minY).takeIf { it > 1e-9 } ?: 1e-9
+            val usableW = (size.width - pad * 2).coerceAtLeast(1f)
+            val usableH = (size.height - pad * 2).coerceAtLeast(1f)
+            fun point(index: Int): Offset {
+                val x = pad + (((xs[index] - minX) / dx).toFloat() * usableW)
+                val y = size.height - pad - (((ys[index] - minY) / dy).toFloat() * usableH)
+                return Offset(x, y)
+            }
+            var path = Path()
+            route.forEachIndexed { i, p ->
+                val o = point(i)
+                if (i == 0 || p.segmentStart) {
+                    if (i > 0) drawPath(path, AccentBlue, style = Stroke(width = 4.dp.toPx(), cap = StrokeCap.Round))
+                    path = Path().apply { moveTo(o.x, o.y) }
+                } else path.lineTo(o.x, o.y)
+            }
+            drawPath(path, AccentBlue, style = Stroke(width = 4.dp.toPx(), cap = StrokeCap.Round))
+            drawCircle(AccentBlue, 5.dp.toPx(), point(0))
+            drawCircle(endpointColor, 6.dp.toPx(), point(route.lastIndex))
+        }
+    }

+    return when { minutes < 1 -> "ahora"; minutes < 60 -> "${minutes}m"; minutes < 1440 -> "${minutes / 60}h"; else -> "${minutes / 1440}d" }
