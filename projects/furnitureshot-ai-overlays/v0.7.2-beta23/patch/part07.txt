+        strength: Float,
+    ): StructureProtectionResult {
+        if (
+            reference.width != candidate.width || reference.height != candidate.height ||
+            alpha.size != reference.width * reference.height
+        ) {
+            val geometry = StructureGeometryLockEngine.evaluate(reference, candidate, alpha)
+            return StructureProtectionResult(
+                candidate,
+                StructureProtectionReport(0f, 0f, geometry, accepted = false),
+            )
+        }
+
+        val width = reference.width
+        val height = reference.height
+        val ref = IntArray(width * height)
+        val cand = IntArray(width * height)
+        reference.getPixels(ref, 0, width, 0, 0, width, height)
+        candidate.getPixels(cand, 0, width, 0, 0, width, height)
+        val out = cand.clone()
+
+        val lock = strength.coerceIn(0f, 1f)
+        var protected = 0L
+        var eligible = 0L
+        var protectionSum = 0f
+
+        for (y in 1 until height - 1) {
+            for (x in 1 until width - 1) {
+                val i = y * width + x
+                val a = alpha[i]
+                if (a < 0.20f) continue
+                eligible++
+
+                val gx = abs(luma(ref[i + 1]) - luma(ref[i - 1]))
+                val gy = abs(luma(ref[i + width]) - luma(ref[i - width]))
+                val edge = max(gx, gy)
+                val alphaEdge = max(
+                    abs(alpha[i + 1] - alpha[i - 1]),
+                    abs(alpha[i + width] - alpha[i - width]),
+                )
+
+                // Strong boundaries are nearly immutable. Medium-frequency surface
+                // detail receives a lighter lock so material recovery can still work.
+                val structural = when {
+                    alphaEdge > 0.18f -> 1f
+                    edge >= 0.095f -> 1f
+                    edge >= 0.060f -> 0.88f
+                    edge >= 0.038f -> 0.64f
+                    edge >= 0.022f -> 0.34f
+                    else -> 0f
+                }
+                if (structural <= 0f) continue
+
+                val protection = (structural * lock * (0.78f + a.coerceIn(0f, 1f) * 0.22f)).coerceIn(0f, 0.995f)
+                if (protection < 0.02f) continue
+                protected++
+                protectionSum += protection
+
+                val r0 = ref[i]
+                val c0 = cand[i]
+                out[i] = Color.argb(
+                    Color.alpha(c0),
+                    mix(Color.red(c0), Color.red(r0), protection),
+                    mix(Color.green(c0), Color.green(r0), protection),
+                    mix(Color.blue(c0), Color.blue(r0), protection),
+                )
+            }
+        }
+
+        val protectedBitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888).also {
+            it.setPixels(out, 0, width, 0, 0, width, height)
+        }
+        val geometry = StructureGeometryLockEngine.evaluate(reference, protectedBitmap, alpha)
+        val ratio = if (eligible == 0L) 0f else protected.toFloat() / eligible.toFloat()
+        val meanProtection = if (protected == 0L) 0f else protectionSum / protected.toFloat()
+        val accepted = geometry.accepted
+        return StructureProtectionResult(
+            protectedBitmap,
+            StructureProtectionReport(ratio, meanProtection, geometry, accepted),
+        )
+    }
+
+    private fun mix(candidate: Int, reference: Int, amount: Float): Int =
+        (candidate * (1f - amount) + reference * amount).roundToInt().coerceIn(0, 255)
+
+    private fun luma(c: Int): Float =
+        (0.2126f * Color.red(c) + 0.7152f * Color.green(c) + 0.0722f * Color.blue(c)) / 255f
+}
